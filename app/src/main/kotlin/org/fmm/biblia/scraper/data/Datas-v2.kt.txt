package org.fmm.biblia.scraper.data

import kotlinx.serialization.Serializable

@Serializable
enum class CategoriaDocumento {
    LIBRO_BIBLICO,
    INTRODUCCION,
    APENDICE
}

@Serializable
data class DocumentoBiblico(
    val id: String,                     // Ej: "libro25sab", "bloque01penta"
    val titulo: String,                 // Ej: "SABIDURÍA", "Introducción al Pentateuco"
    val categoria: CategoriaDocumento,
    val orden: Int,
    val urlOrigen: String,
    val bloques: List<BloqueContenido> // Secuencia lineal y ordenada del contenido
)

// === BLOQUES DE CONTENIDO (Todos al mismo nivel de abstracción) ===

@Serializable
sealed interface BloqueContenido

/**
 * Representa cualquier título o encabezado, sin importar si es de un libro,
 * una introducción o una perícopa dentro de un capítulo.
 */
@Serializable
data class Titulo(
    val nivel: Int,                     // 1: Título principal / Sección, 2: Subtítulo / Perícopa, 3: Sub-encabezado
    val texto: String,
    val paralelos: List<String> = emptyList(),
    val llamadasNotas: List<String> = emptyList()
) : BloqueContenido

/**
 * Representa párrafos de texto continuo (prosa), característicos de las
 * Introducciones y Apéndices.
 */
@Serializable
data class Parrafo(
    val texto: String,
    val llamadasNotas: List<String> = emptyList()
) : BloqueContenido

/**
 * Representa un versículo bíblico numerado.
 * Lleva la información del capítulo al que pertenece para no requerir contenedores anidados.
 */
@Serializable
data class Versiculo(
    val capitulo: Int,
    val numero: Int,
    val texto: String,
    val esPoesia: Boolean = false,
    val paralelos: List<String> = emptyList(),
    val llamadasNotas: List<String> = emptyList()
) : BloqueContenido

/**
 * Representa las notas al pie asociadas directamente al bloque o pasaje correspondiente.
 */
@Serializable
data class NotaAlPie(
    val id: String,                     // Ej: "a", "n1"
    val texto: String,
    val referencia: String? = null      // Ej: "1,1" o capítulo/versículo asociado
) : BloqueContenido
